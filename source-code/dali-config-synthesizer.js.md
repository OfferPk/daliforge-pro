class DALIConfigSynthesizer {  
  generateNftables(config \= {}) {  
    const bmsIp \= config.bmsIp \|\| '192.168.20.10';  
    const gwSubnet \= config.gwSubnet \|\| '192.168.20.0/24';

    return \`\#\!/usr/sbin/nft \-f  
\# DALIForge Pro \- Linux nftables Smart Lighting Conduit Firewall  
\# Hardened Ruleset for DALI-to-BACnet / DALI-to-KNX / DALI-over-IP Gateways  
\# Reference: IEC 62443-3-3 Zone & Conduit Security

flush ruleset

table inet dali\_firewall {  
    \# Authorized Building Management Systems (BMS / Tridium Niagara / Desigo)  
    set authorized\_bms {  
        type ipv4\_addr  
        elements \= { \${bmsIp} }  
    }

    \# Dedicated DALI Lighting Gateway Subnet  
    set dali\_gateways {  
        type ipv4\_addr  
        flags interval  
        elements \= { \${gwSubnet} }  
    }

    chain input {  
        type filter hook input priority filter; policy drop;

        iif "lo" accept  
        ct state established,related accept  
        ct state invalid drop

        ip protocol icmp limit rate 5/second accept  
    }

    chain forward {  
        type filter hook forward priority filter; policy drop;

        ct state established,related accept  
        ct state invalid drop

        \# 1\. Allow BACnet/IP (UDP Port 47808\) between BMS and DALI Gateways  
        ip saddr @authorized\_bms ip daddr @dali\_gateways udp dport 47808 accept  
        ip saddr @dali\_gateways ip daddr @authorized\_bms udp sport 47808 accept

        \# 2\. Allow DALI-over-IP Tunneling (UDP Port 51820\) strictly within Lighting Subnet  
        ip saddr @dali\_gateways ip daddr @dali\_gateways udp dport 51820 accept

        \# 3\. Allow Secure Commissioning HTTPS (TCP Port 443\) strictly from authorized BMS station  
        ip saddr @authorized\_bms ip daddr @dali\_gateways tcp dport 443 ct state new limit rate 15/minute accept  
        ip saddr @authorized\_bms ip daddr @dali\_gateways tcp dport 443 accept

        \# Log unauthorized attempts to manipulate lighting gateways  
        udp dport 47808 log prefix "\[DALI-UNAUTH-BACNET\] " drop  
        udp dport 51820 log prefix "\[DALI-UNAUTH-UDP-IP\] " drop  
        tcp dport 443 log prefix "\[DALI-UNAUTH-HTTPS\] " drop  
    }

    chain output {  
        type filter hook output priority filter; policy accept;  
    }  
}  
\`;  
  }

  generateSuricataRules(config \= {}) {  
    const sidBase \= config.sidBase \|\| 20261710;  
    return \`\# Suricata L7 Industrial IDS Rules for DALI-over-IP & BACnet Lighting Gateways  
\# Generated by DALIForge Pro \- v1.0.0  
\# Target: DALI-over-IP (UDP Port 51820\) & BACnet Lighting Interfaces (UDP Port 47808\)

\# 1\. Detect DALI Broadcast Turn-Off / Blackout Command (0xFE 0x00 / 0xFF 0x00)  
alert udp any any \-\> any 51820 (msg:"DALI-IDS CRITICAL \- Broadcast Blackout Frame (DAPC 0 / OFF) Injected"; content:"\|fe 00\|"; offset:0; depth:4; priority:1; classtype:attempted-dos; sid:\${sidBase \+ 1}; rev:1;)

\# 2\. Detect DALI INITIALIZE Re-Addressing Probe (Special Command 0xA1)  
alert udp any any \-\> any 51820 (msg:"DALI-IDS Unauthorized INITIALIZE Commissioning Command (0xA1)"; content:"\|a1\|"; offset:0; depth:2; classtype:attempted-admin; sid:\${sidBase \+ 2}; rev:1;)

\# 3\. Detect Emergency Lighting Function Test Trigger (Command 240 / 0xF0)  
alert udp any any \-\> any 51820 (msg:"DALI-IDS Emergency Luminaire Function Test Initiated (Cmd 240)"; content:"\|f0\|"; offset:1; depth:2; classtype:protocol-command-decode; sid:\${sidBase \+ 3}; rev:1;)

\# 4\. Detect Rapid Stroboscopic Cycling / Flooding on DALI Gateway  
alert udp any any \-\> any 51820 (msg:"DALI-IDS High-Frequency DAPC Level Cycling (Stroboscopic Hazard)"; threshold:type threshold, track by\_src, count 25, seconds 2; classtype:bad-unknown; sid:\${sidBase \+ 4}; rev:1;)  
\`;  
  }  
}

if (typeof module \!== 'undefined' && module.exports) {
